
NIAT Kit niat/ Library — Module Pattern
================================================================================

PATTERN TO FOLLOW FOR ALL NEW MODULES:
--------------------------------------------------------------------------------
1. Module docstring at top describing purpose
2. Hardware imports at top (hidden from students)
3. Import pin constants from niat._board
4. Constants/lookup tables if needed (e.g., COLORS dict)
5. Single class named after physical object
6. __init__ initializes hardware with try/except for friendly errors
7. Methods named as plain English actions
8. All methods have docstrings with Args/Returns
9. Returns real units always (Celsius, percent, lux, etc.)
10. Friendly error messages via niat._errors

"""
NIAT Kit LED Module
Controls the onboard NeoPixel LED.
"""

import board
import neopixel
import time


# Color name to RGB value mapping
COLORS = {
    "red":     (255, 0, 0),
    "green":   (0, 255, 0),
    "blue":    (0, 0, 255),
    "white":   (255, 255, 255),s
    "yellow":  (255, 255, 0),
    "orange":  (255, 165, 0),
    "purple":  (128, 0, 128),
    "cyan":    (0, 255, 255),
    "pink":    (255, 105, 180),
    "off":     (0, 0, 0),
}


class LED:
    """
    A class to control the onboard NeoPixel LED.
    
    Provides simple methods to turn the LED on/off, change colors,
    and create blinking patterns.
    """
    
    def __init__(self):
        """
        Initialize the LED.
        
        The LED starts in the off state.
        """
        try:
            self._pixel = neopixel.NeoPixel(board.NEOPIXEL, 1, brightness=0.3)
            self._pixel[0] = (0, 0, 0)
        except Exception:
            raise RuntimeError(
                "LED not found. Is the NeoPixel connected?"
            )
    
    def on(self, color="white"):
        """
        Turn the LED on with the specified color.
        
        Args:
            color: Color name or RGB tuple (r, g, b)
        """
        if isinstance(color, str):
            rgb = COLORS.get(color.lower(), (255, 255, 255))
        else:
            rgb = color
        self._pixel[0] = rgb
    
    def off(self):
        """
        Turn the LED off.
        """
        self._pixel[0] = (0, 0, 0)
    
    def is_on(self):
        """
        Check if the LED is currently on.
        
        Returns:
            True if the LED is on, False otherwise
        """
        return self._pixel[0] != (0, 0, 0)
    
    def blink(self, color="white", speed=0.5, count=3):
        """
        Blink the LED a specified number of times.
        
        Args:
            color: Color name or RGB tuple
            speed: Time in seconds for each on/off cycle
            count: Number of times to blink
        """
        for _ in range(count):
            self.on(color)
            time.sleep(speed)
            self.off()
            time.sleep(speed)
